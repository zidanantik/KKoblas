if (g) g.style.display = 'none'; showStatus('✓ Dárkový poukaz na ' + (pkgTitles[targetPkg] || targetPkg.toUpperCase()) + ' aktivován (100% sleva).', 'ok'); var targetEl = document.getElementById('card-' + targetPkg);
